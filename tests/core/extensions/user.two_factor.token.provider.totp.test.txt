import { describe, it, expect, beforeEach } from "vitest";
import { IdentityUser } from "../../../src/core/types/index.js";
import { UserManager } from "../../../src/core/identity/index.js";
import { AuthenticatorTokenProvider, Rfc6238AuthenticationService } from "../../../src/core/extensions/index.js";

// Simple test user
class TestUser extends IdentityUser<string> {
  constructor(userName: string) {
    super(userName);
  }
}

// Fake manager with minimal implementation
class FakeUserManager extends UserManager<string, TestUser> {
  constructor() {
    // pass stubs for constructor args
    super({} as any, {} as any, {} as any, [] as any, [] as any, {} as any, {} as any);
  }

  async createSecurityTokenAsync(user: TestUser): Promise<Uint8Array> {
    // return deterministic token
    return Buffer.from("secret");
  }

  async getUserIdAsync(user: TestUser): Promise<string> {
    return user.id;
  }
}

describe("TotpSecurityStampBasedTokenProvider", () => {
  let provider: AuthenticatorTokenProvider<string, TestUser>;
  let manager: FakeUserManager;
  let user: TestUser;

  beforeEach(() => {
    provider = new AuthenticatorTokenProvider<string, TestUser>();
    manager = new FakeUserManager();
    user = new TestUser("user");
    user.id = "123";
  });

  it("generates a 6-digit code", async () => {
    const code = await provider.generateAsync("login", manager, user);
    expect(code).toMatch(/^\d{6}$/);
  });

  it("validates a correct code", async () => {
    const code = await provider.generateAsync("login", manager, user);
    const valid = await provider.validateAsync("login", code, manager, user);
    expect(valid).toBe(true);
  });

  it("rejects non-numeric token", async () => {
    const valid = await provider.validateAsync("login", "abc", manager, user);
    expect(valid).toBe(false);
  });

  it("rejects incorrect code", async () => {
    const valid = await provider.validateAsync("login", "000000", manager, user);
    expect(valid).toBe(false);
  });

  it("returns modifier string correctly", async () => {
    const modifier = await provider.getUserModifierAsync("login", manager, user);
    expect(modifier).toBe("Totp:login:123");
  });

  it("canGenerateTwoFactorTokenAsync returns true when token exists", async () => {
    const result = await provider.canGenerateTwoFactorTokenAsync(manager, user);
    expect(result).toBe(true);
  });
});

describe("Rfc6238AuthenticationService", () => {
  it("generates deterministic code for same input", () => {
    const token = Buffer.from("secret");
    const modifier = "Totp:login:123";
    const code1 = Rfc6238AuthenticationService.generateCode(token, modifier);
    const code2 = Rfc6238AuthenticationService.generateCode(token, modifier);
    expect(code1).toBe(code2);
  });

  it("validateCode returns true for matching code", () => {
    const token = Buffer.from("secret");
    const modifier = "Totp:login:123";
    const code = Rfc6238AuthenticationService.generateCode(token, modifier);
    const valid = Rfc6238AuthenticationService.validateCode(token, code, modifier);
    expect(valid).toBe(true);
  });

  it("validateCode returns false for non-matching code", () => {
    const token = Buffer.from("secret");
    const modifier = "Totp:login:123";
    const valid = Rfc6238AuthenticationService.validateCode(token, 111111, modifier);
    expect(valid).toBe(false);
  });
});
